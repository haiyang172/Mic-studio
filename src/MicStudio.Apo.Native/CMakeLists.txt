cmake_minimum_required(VERSION 3.21)
project(MicStudioNativeApo VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native APO requires the reviewed Windows x64 MSVC/Windows SDK toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the prototype self-contained; formal driver-package policy is reviewed
# separately before signed system deployment. No VC redistributable is installed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MicStudioApo SHARED MicStudioApo.cpp MicStudioApo.def)
target_compile_definitions(MicStudioApo PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(MicStudioApo PRIVATE /W4 /permissive- /sdl /guard:cf /EHsc)
target_link_libraries(MicStudioApo PRIVATE ole32 propsys uuid)
target_link_options(MicStudioApo PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /MANIFEST:NO)
set_target_properties(MicStudioApo PROPERTIES OUTPUT_NAME MicStudioApo)
enable_testing()
add_executable(MicStudioApoNativeTests ../../tests/MicStudio.Apo.Native.Tests/NativeTests.cpp)
target_compile_definitions(MicStudioApoNativeTests PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(MicStudioApoNativeTests PRIVATE /W4 /permissive- /sdl /EHsc)
target_include_directories(MicStudioApoNativeTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(MicStudioApoNativeTests PRIVATE ole32 propsys uuid)
add_dependencies(MicStudioApoNativeTests MicStudioApo)
add_test(NAME NativeApoOffline COMMAND MicStudioApoNativeTests $<TARGET_FILE:MicStudioApo>)
